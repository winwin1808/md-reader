```tsx
// packages/ui/src/components/Editor.tsx
import type * as Y from 'yjs'
import type { EditorView } from '@codemirror/view'

export interface EditorProps {
  yText: Y.Text
  undoManager?: Y.UndoManager
  className?: string
  placeholder?: string
  readOnly?: boolean
  onScroll?: (scrollTop: number, scrollHeight: number, clientHeight: number) => void
  onEditorReady?: (view: EditorView) => void
}

export function Editor(props: EditorProps): React.JSX.Element
```
INVARIANTS
- Packages boundary: `packages/ui` is pure React & CSS (NO `chrome.*`), pure presentation.
- Yjs binding: Must bind `yText` with `yCollab(props.yText, null, { undoManager: props.undoManager })`. Changes in EditorView dispatch to Y.Text; changes to Y.Text reflect in EditorView without cursor jumping.
- CodeMirror 6 setup:
  - Extensions: `markdown()`, `lineNumbers()`, `highlightActiveLine()`, `bracketMatching()`, `syntaxHighlighting(defaultHighlightStyle)`.
  - DOM scroll listener: Report `onScroll` when user scrolls editor container (`domEventHandlers({ scroll: ... })`).
  - Read-only handling: `EditorState.readOnly.of(Boolean(props.readOnly))`.
- Lifecycle & Memory: Create `EditorView` on container mount in `useEffect`; call `view.destroy()` on unmount.
- Styling in `packages/ui/src/theme/editor.css`:
  - `.cm-editor` container styled with theme-aware font (`font-family: var(--md-font-mono, ui-monospace, monospace)`), line height, height 100%, overflow auto.
  - Border, background (`var(--md-bg-editor, var(--md-bg, #ffffff))`), text color (`var(--md-text, #1e293b)`).
  - Import `editor.css` in `packages/ui/src/viewer.css`.
- In `packages/ui/src/index.ts`:
  - Export `Editor`, `EditorProps`.
