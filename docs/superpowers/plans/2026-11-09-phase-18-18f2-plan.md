            }
            export const FEEDBACK_EMAIL = 'qhao.work@gmail.com'
            export const HELP_URL = 'https://winwin1808.github.io/md-reader/'
            export const PRIVACY_URL = 'https://winwin1808.github.io/md-reader/privacy.html'
            export function browserLabel(nav?: { userAgent?: string; userAgentData?: { brands?: { brand: string; version: string }[] } }): string
            export function diagnostics(): { version: string; browser: string; locale: string }
            export function sendFeedback(input: FeedbackInput, fetchImpl?: typeof fetch, base?: string): Promise<'sent' | 'failed'>
            export function feedbackMailto(d: { kind: 'bug' | 'idea'; message: string; email: string }, diag: { version: string; browser: string; locale: string }): string
            export function toFeedbackInput(d: { kind: 'bug' | 'idea'; message: string; email: string }, diag: { version: string; browser: string; locale: string }): FeedbackInput
            ```
            `browserLabel`: brand "Google Chrome" / "Microsoft Edge" / "Chromium" từ `userAgentData.brands` → `"<brand ngắn: Chrome|Edge|Chromium> <major>"`; không có thì regex `Edg\/(\d+)` → `Edge N`, `Firefox\/(\d+)` → `Firefox N`, `Chrome\/(\d+)` → `Chrome N`; không ra → `'unknown'`. Chỉ tên + số phiên bản chính, không hệ điều hành, không chuỗi UA đầy đủ.
            `diagnostics()`: `version` = `chrome.runtime?.getManifest?.().version ?? 'unknown'`; `browser` = `browserLabel(navigator)`; `locale` = `chrome.i18n?.getUILanguage?.()` bắt đầu bằng `vi` → `'vi'`, ngược lại `'en'`.
            `sendFeedback`: `base` mặc định `DEFAULT_SERVER_URL` (từ `../shared/collab`), bỏ `/` cuối; rỗng → `'failed'` không gọi fetch. `fetchImpl(base + '/api/feedback', { method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify({ ...input, website: '' }) })`; `res.status === 204` hoặc `res.ok` → `'sent'`; khác (gồm 429, 400, 5xx) hoặc ném lỗi mạng → `'failed'`. Không gửi header nào khác (không `x-mdr-install`, không cookie: `credentials: 'omit'`).
            `toFeedbackInput`: `{ kind, reason: null, message: d.message, email: d.email.trim() || null, version: diag.version, browser: diag.browser, locale: diag.locale === 'vi' ? 'vi' : 'en' }`.
            `feedbackMailto`: `mailto:${FEEDBACK_EMAIL}?subject=<encodeURIComponent("MD Reader feedback (" + kind + ")")>&body=<encodeURIComponent(message + "\n\n---\nVersion: … · Browser: … · Language: …")>`.
INVARIANTS  `apps/extension` là lớp duy nhất dùng `chrome.*`; gọi qua optional chaining.
            Không thu telemetry: chỉ gửi đúng các trường của form + version, browser (tên + số chính), locale; không gửi `x-mdr-install`, nội dung tài liệu hay URL.
            Không import từ `apps/api` (khai lại kiểu).
