                // parse with new URL (invalid → false); protocol http:, https: or file:;
                // pathname (decoded, case-insensitive) ends with '.' + one of MD_EXTENSIONS;
                // query and hash ignored; github.com/<o>/<r>/blob/… → false (GitHub renders it).
              export function isStaleCacheEntry(value: unknown, now: number, maxAgeMs = 600_000): boolean
                // string (legacy) → true; { text: string; ts: number } with now - ts > maxAgeMs → true; else false.
            manifest.json: "permissions": ["storage", "activeTab"] (drop "tabs", "scripting" and
              anything else not needed — keep "storage"/"activeTab" only unless the worker code you
              keep needs another; say so in NOTES); host_permissions and content_scripts[].matches =
              for each ext in MD_EXTENSIONS both `*://*/*.<ext>` and `*://*/*.<EXT>` (upper case),

              "web_accessible_resources": [{ "resources": ["app.css"], "matches": <the same list> }].

              click handler decides with isMarkdownUrl(tab.url); on runtime.onStartup and
              runtime.onInstalled: chrome.storage.local.get(null) → remove every key starting with
              'file_content_' whose value isStaleCacheEntry(value, Date.now()).
