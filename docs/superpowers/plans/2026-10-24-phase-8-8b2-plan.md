```ts
import type { ResolveTokenResponse } from '@md-reader/core'
export function tokenFromPath(pathname: string): string | null          // /s/<token> với token /^[A-Za-z0-9_-]{16,}$/, else null
export type ShareState =
  | { kind: 'loading' }
  | { kind: 'ready'; info: ResolveTokenResponse; text: string }
  | { kind: 'gone' }                                                     // 410
  | { kind: 'error'; message: string }                                   // mạng / 5xx / snapshot lỗi
export async function loadShare(token: string, fetchFn: typeof fetch = fetch): Promise<ShareState>
  // GET API_PATHS.resolve(token) → 410 → gone; !ok → error; ok → GET API_PATHS.snapshot(token) → arrayBuffer → new Y.Doc({gc:false}) + Y.applyUpdate → text = exportText(doc) → ready
export function renderShareHtml(text: string): string                   // core parse + render với tuỳ chọn mặc định của app page (gfm, math, alerts) → HTML đã sanitize cho Viewer
```
