INTERFACES  export type IconName = 'book' | 'edit' | 'split' | 'present' | 'mindmap' | 'export' | 'settings' | 'search' | 'close'
              | 'chevron-left' | 'chevron-right' | 'chevron-down' | 'more' | 'open-file' | 'save' | 'check' | 'alert' | 'fullscreen'
              | 'zoom-in' | 'zoom-out' | 'expand-all' | 'collapse-all' | 'copy' | 'speaker' | 'toc' | 'font-smaller' | 'font-larger'
              | 'undo' | 'redo' | 'menu' | 'share'
            export function Icon(props: { name: IconName; size?: 16 | 20 | 24; label?: string; className?: string }): JSX.Element
              <svg viewBox="0 0 24 24" width/height = size ?? 20, fill="none", stroke="currentColor", stroke-width="1.6",
              stroke-linecap="round", stroke-linejoin="round", focusable="false">; label → role="img" + aria-label; no label → aria-hidden="true".
              Paths live in one `const PATHS: Record<IconName, string[]>` (each entry = list of `d` strings; circles as path arcs).
            Drawing rules (24 grid, 2px padding, optical size ~18px, no fills, no brand shapes): book = open book with spine;
              edit = pencil at 45°; split = rectangle with a vertical middle line; present = easel/screen on a stand; mindmap =
              node with three branches; export = tray with an arrow out/up; settings = gear (6–8 teeth) or three sliders;
              search = circle + handle; close = X; chevrons = single angle; more = three dots horizontal; open-file = folder;
              save = arrow down into a tray (not a floppy); check = tick; alert = triangle with !; fullscreen = four corners;
              zoom-in/out = magnifier with +/−; expand-all/collapse-all = stacked chevrons apart/together; copy = two offset
              rectangles; speaker = person at a lectern or monitor with notes lines; toc = three lines with bullets; font-smaller
              / font-larger = "A" with −/+ ; undo/redo = curved arrow; menu = three lines; share = three connected nodes.
            TopToolbar, Deck, MindmapModal: replace EVERY emoji / Unicode pictograph / arrow glyph used as an icon with <Icon>.
              Icon-only buttons get a Vietnamese aria-label and title (e.g. "Tìm trong tài liệu", "Trình chiếu", "Sơ đồ tư duy",
