```ts
// packages/core/src/slides/types.ts
export interface Slide {
  id: string
  index: number
  rawMarkdown: string
  html: string
  notes: string[]
  title?: string
}

export interface SlideSplitOptions {
  strategy?: 'hr' | 'heading' | 'auto' // default 'auto'
  headingDepth?: number // default 2
}

// packages/core/src/slides/split.ts
export function splitIntoSlides(source: string, options?: SlideSplitOptions): Slide[]
```
INVARIANTS
- `packages/core` is pure TypeScript. NO React, NO `chrome.*`, NO DOM APIs (`document`, `window`).
- Use existing `renderToHtml` from `@md-reader/core` to render slide markdown to sanitized HTML string.
- Speaker notes extraction: match HTML comments `<!-- note: ... -->` or `<!-- speaker: ... -->` (case-insensitive, multiline). Extract notes into `slide.notes: string[]` (trimmed).
- Clean HTML output: Strip `<!-- note: ... -->` and `<!-- speaker: ... -->` from the markdown before calling `renderToHtml` so notes NEVER appear in `slide.html`.
- Splitting logic:
  - Default strategy `'auto'`: if source contains `---` (as a markdown hr, e.g. line with `^---\\s*$` not in code block), split on `---`. If not, split on headings with depth <= `options?.headingDepth ?? 2`.
  - Fallback: if source has neither `---` nor matching headings, the whole source is 1 slide.
  - If source is empty, return 1 empty slide `{ id: 'slide-0', index: 0, rawMarkdown: '', html: '', notes: [] }`.
- Export `Slide`, `SlideSplitOptions`, `splitIntoSlides` from `packages/core/src/slides/index.ts` and `packages/core/src/index.ts`.
