```ts
import type { ShareRole, ShareLink, CreateRoomResponse, CreateLinkRequest } from '@md-reader/core'
export class RoomNotFoundError extends Error {}
export interface TokenStore {
  createRoom(): Promise<CreateRoomResponse>
  verifyOwner(roomId: string, ownerToken: string): Promise<boolean>
  createLink(roomId: string, req: CreateLinkRequest): Promise<ShareLink>     // throws RoomNotFoundError; ShareLink.token = raw token (lần duy nhất)
  listLinks(roomId: string): Promise<ShareLink[]>                             // token = '' ; sắp xếp createdAt giảm dần; gồm cả link đã thu hồi/hết hạn
  revoke(roomId: string, tokenPrefix: string): Promise<boolean>              // true nếu có link khớp và chưa revoked
  resolve(token: string): Promise<{ roomId: string; role: ShareRole } | null> // null = không tồn tại / hết hạn / đã thu hồi; owner token → role 'edit'
}
export async function hashToken(token: string): Promise<string>   // SHA-256 hex (crypto.subtle)
export function newToken(): string                                // 32 byte ngẫu nhiên, base64url không padding (43 ký tự)
export function tokenPrefix(token: string): string                // 6 ký tự đầu
export function newRoomId(): string                               // 16 byte base64url (22 ký tự)
export class MemoryTokenStore implements TokenStore { constructor(now: () => number = Date.now) }
```
            SQL (`migrations/0001_init.sql`), đúng schema sau:
```sql
CREATE TABLE rooms (id TEXT PRIMARY KEY, owner_token_hash TEXT NOT NULL UNIQUE, title TEXT, created_at INTEGER NOT NULL);
CREATE TABLE links (token_hash TEXT PRIMARY KEY, token_prefix TEXT NOT NULL, room_id TEXT NOT NULL REFERENCES rooms(id),
  role TEXT NOT NULL CHECK (role IN ('view','edit')), label TEXT NOT NULL DEFAULT '', created_at INTEGER NOT NULL,
  expires_at INTEGER, revoked_at INTEGER);
CREATE INDEX links_room ON links(room_id);
```
