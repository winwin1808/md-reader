```ts
// @md-reader/ui
import type { TocEntry } from '@md-reader/core'
import React from 'react'

export interface ViewerProps {
  /** HTML đã sanitize từ core/render. Viewer KHÔNG tự sanitize lại. */
  html: string
  className?: string
  /** Gọi khi heading đang đọc đổi, để Toc highlight. */
  onActiveHeadingChange?: (id: string | null) => void
}

export function Viewer(props: ViewerProps): React.JSX.Element

export interface TocProps {
  entries: TocEntry[]
  activeId?: string | null
  onSelect?: (id: string) => void
}

export function Toc(props: TocProps): React.JSX.Element | null
```
INVARIANTS  
- `packages/ui`: không `chrome.*`. Nhận dữ liệu qua props. Không component nào tự đọc file hay biết provider.
- `Viewer` dùng `dangerouslySetInnerHTML`. Bắt buộc ghi comment giải thích đây là HTML đã qua `sanitizeSchema` từ core.
- `onActiveHeadingChange` dùng `IntersectionObserver` trên các heading (`h1`-`h6`) có `id`. Nếu `IntersectionObserver` không tồn tại (như trong môi trường jsdom không có polyfill), bỏ qua (không throw).
- `Toc` render `<nav>`, lồng nhau theo `children` của `entries`. Nếu `entries: []` hoặc mảng rỗng thì return `null` (không render `<nav>` rỗng). `activeId` thì render `aria-current="true"`. Nhớ style đơn giản hoặc class.
- `viewer.css` sử dụng các token CSS (vd `--md-bg`, `--md-fg`) cho typography và map `.hljs-*` sang các màu `--md-hljs-*`. File tsx nhớ `import './viewer.css'`.
- `index.ts` xuất `Viewer` và `Toc`.
